local function getCatchTarget(char)

local hrp =

char:FindFirstChild("HumanoidRootPart")



if not hrp then

return nil

end



local look = hrp.CFrame.LookVector



local offset =

look * 0.4

+ Vector3.new(0, 0.2, 0)



local right = armPos(char, {

"RightHand",

"RightLowerArm",

"RightUpperArm",

"Right Arm",

"RightArm",

})



local left = armPos(char, {

"LeftHand",

"LeftLowerArm",

"LeftUpperArm",

"Left Arm",

"LeftArm",

})



if right then

right += offset

end



if left then

left += offset

end



if right and left and catchBall then

if (catchBall.Position - left).Magnitude

< (catchBall.Position - right).Magnitude then

return left

end



return right

end



return right

or left

or hrp.Position

+ Vector3.new(0, 0.75, 0)

+ look * 0.55

end



local function isBallOnMe(ball, char)

if not ball or not char then

return false

end



if ball:IsDescendantOf(char) then

return true

end



for _, w in ipairs(ball:GetChildren()) do

-- boost



local function isOnTopOfPart(part)

if not rootPart

or not part

or not part:IsA("BasePart") then

return false

end



local relative =

part.CFrame:PointToObjectSpace(

rootPart.Position

)



local halfX =

part.Size.X / 2



local halfZ =

part.Size.Z / 2



if math.abs(relative.X) > halfX + 1 then

return false

end



if math.abs(relative.Z) > halfZ + 1 then

return false

end



local topY =

part.Position.Y

+ part.Size.Y / 2



local rootBottomY =

rootPart.Position.Y - 2.5



local verticalDistance =

rootBottomY - topY



return verticalDistance >= -1.5

and verticalDistance <= TOP_CONTACT_TOLERANCE

end



local function isOnTopOfPlayer(other)

local targetCharacter = other.Character



if not targetCharacter then

return false

end



local head =

targetCharacter:FindFirstChild("Head")



if head and isOnTopOfPart(head) then

return true

end



local upperTorso =

targetCharacter:FindFirstChild("UpperTorso")



if upperTorso and isOnTopOfPart(upperTorso) then

return true

end



return false

end
